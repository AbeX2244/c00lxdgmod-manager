# --- Optional: fast GMA extraction ---
# C++ backend for reading .gma files. Falls back to the pure-Python
# parser (PureGMA in core.py) if not installed.
# Requires Python 3.11+ on Linux/macOS, 3.10+ on Windows.
sourcepp>=0.2.0

# --- Optional: AES-encrypted ZIP support ---
# Enables reading ZIP archives with WinZip AES-128/256 encryption.
# Without it, only legacy ZipCrypto or unencrypted ZIPs work.
# Pulls pycryptodomex automatically.
pyzipper>=0.3.6

# --- Optional: RAR support (not yet wired into the UI) ---
